import base64,os
FS='node_modules/@fontsource/'
faces=[('Chakra Petch','chakra-petch',[500,600,700]),('IBM Plex Sans','ibm-plex-sans',[400,500,600]),('IBM Plex Mono','ibm-plex-mono',[400,500])]
ff=''
for fam,d,ws in faces:
    for w in ws:
        b=base64.b64encode(open(f'{FS}{d}/files/{d}-latin-{w}-normal.woff2','rb').read()).decode()
        ff+=f'@font-face{{font-family:"{fam}";font-weight:{w};src:url(data:font/woff2;base64,{b}) format("woff2")}}\n'
css=open('guide.css').read();body=open('guide_body.html').read()
import json,os,re
if os.path.exists('zooms.json'):
    zs=json.load(open('zooms.json'));it=iter(zs)
    body=re.sub(r'<section class="page ([^"]*)" style="([^"]*)"',lambda m:'<section class="page %s" style="%s;--z:%s"'%(m.group(1),m.group(2),next(it)),body)
    css+='\n.page:not(.cover)>*:not(.pf){zoom:var(--z,1)}'
T='<title>Warzone Shader Guide</title>\n'
gf='<link rel="preconnect" href="https://fonts.googleapis.com"><link rel="preconnect" href="https://fonts.gstatic.com" crossorigin><link href="https://fonts.googleapis.com/css2?family=Chakra+Petch:wght@500;600;700&family=IBM+Plex+Mono:wght@400;500&family=IBM+Plex+Sans:wght@400;500;600&display=swap" rel="stylesheet">\n'
open('index.html','w').write(T+gf+'<style>'+css+'</style>\n<div class="doc">'+body+'</div>')
open('local.html','w').write('<!doctype html><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1">'+T+'<style>'+ff+css+'</style><div class="doc">'+body+'</div>')
